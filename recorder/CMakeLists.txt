# hepsimlib-recorder: a standalone Geant4 application (docs/recorder.md). Needs Geant4 >= 11.2 built with GDML.
cmake_minimum_required(VERSION 3.16)
project(hepsimlib-recorder CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Geant4 11.2 REQUIRED gdml)
include(${Geant4_USE_FILE})
add_executable(hepsimlib-recorder src/main.cc src/Config.cc src/Capture.cc src/Output.cc)
target_include_directories(hepsimlib-recorder PRIVATE src)
target_link_libraries(hepsimlib-recorder PRIVATE ${Geant4_LIBRARIES})
target_compile_options(hepsimlib-recorder PRIVATE -Wall -Wextra)
find_package(ZLIB REQUIRED)
set(HEPSIMLIB_CPP ${CMAKE_CURRENT_SOURCE_DIR}/../cpp/include)
# the C++ bank host (docs/geant4-host.md): hepsimlib-hybrid (Geant4), hepsimlib-bank and hepsimlib-replay (no Geant4)
add_executable(hepsimlib-hybrid src/hybrid.cc src/Output.cc)
target_include_directories(hepsimlib-hybrid PRIVATE src ${HEPSIMLIB_CPP})
target_link_libraries(hepsimlib-hybrid PRIVATE ${Geant4_LIBRARIES} ZLIB::ZLIB)
target_compile_options(hepsimlib-hybrid PRIVATE -Wall -Wextra)
foreach(tool hepsimlib-bank hepsimlib-replay)
  add_executable(${tool} ../cpp/tools/bank_tools.cc)
  target_include_directories(${tool} PRIVATE ${HEPSIMLIB_CPP})
  target_link_libraries(${tool} PRIVATE ZLIB::ZLIB)
  target_compile_options(${tool} PRIVATE -Wall -Wextra)
endforeach()
install(TARGETS hepsimlib-recorder hepsimlib-hybrid hepsimlib-bank hepsimlib-replay RUNTIME DESTINATION bin)
